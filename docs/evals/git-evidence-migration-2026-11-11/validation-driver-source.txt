import fs from 'node:fs';
import path from 'node:path';
import {spawnSync} from 'node:child_process';
const root=process.cwd(),out=path.join(root,'docs/evals/git-evidence-migration-2026-11-11');
fs.mkdirSync(out,{recursive:true});
const commands={
 facts:['--import','tsx','tests/test_workflow_facts.ts'],
 design:['--import','tsx','tests/test_design_consolidation.ts'],
 identity:['--import','tsx','tests/test_functional_checker_identity.ts'],
 locks:['--import','tsx','tests/test_owned_lock.ts'],
 persistence:['tests/test_progress_persistence.mjs'],
 native:['tests/test_native_tool_channel.mjs'],
 approval:['--import','tsx','tests/test_native_approval_channel.ts'],
 qualification:['--import','tsx','tests/test_native_channel_qualification.ts'],
 recovery:['--import','tsx','tests/test_run_recovery.ts'],
 faults:['--import','tsx','tests/test_recovery_faults.ts'],
 mermaid:['--import','tsx','tests/test_mermaid_static_pass.ts'],
 closure:['--import','tsx','tests/test_checker_import_closure.ts'],
 typecheck:['node_modules/typescript/bin/tsc','--noEmit'],
 build:['--import','tsx','.artifacts/build-retain-root.ts','--all'],
 distribution:['--import','tsx','tests/test_autonomous_distribution.ts'],
};
for(const name of process.argv.slice(2)){
 if(!commands[name])throw Error('unknown validation group');
 const start=performance.now(),r=spawnSync(process.execPath,commands[name],{cwd:root,encoding:'utf8',timeout:180000,windowsHide:true,env:{...process.env,AIDLC_EVAL_DIR:out}});
 const row={name,argv:[process.execPath,...commands[name]],duration_ms:performance.now()-start,exit_code:r.status,error:r.error?.message||null,signal:r.signal,real_business_run:false};
 fs.writeFileSync(path.join(out,name+'.stdout.txt'),r.stdout||'');fs.writeFileSync(path.join(out,name+'.stderr.txt'),r.stderr||'');fs.writeFileSync(path.join(out,name+'.json'),JSON.stringify(row,null,2)+'\n');console.log(JSON.stringify(row));
 if(r.status!==0){console.error((r.stdout||'')+(r.stderr||''));process.exitCode=1;break;}
}
